sudo apt install -y qemu-utils
qemu-img convert -f qcow2 -O raw ubuntu-20.04-server-cloudimg-amd64.qcow2 disk.raw




mkdir seed
cat > seed/user-data <<'EOF'
#cloud-config
users:
  - name: tony
    sudo: ALL=(ALL) NOPASSWD:ALL
    groups: users, admin
    shell: /bin/bash
    lock_passwd: false
    passwd: $6$rounds=4096$SALT$HASHED_PASSWORD
ssh_pwauth: true
EOF

cat > seed/meta-data <<'EOF'
instance-id: iid-local01
local-hostname: rustvmm
EOF

# 生成 seed.iso
genisoimage -output seed.iso -volid cidata -joliet -rock seed/user-data seed/meta-data
